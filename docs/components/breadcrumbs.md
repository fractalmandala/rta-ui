# Breadcrumbs

Show where a page sits in the site hierarchy with links back up.

Port of `moduix/packages/svelte/src/components/breadcrumbs` with the same public parts and a byte-identical CSS Module, rebuilt on `src/lib/internal` instead of Ark UI.

## Installation

```sh
pnpm add rta-ui
```

`rta-ui` ships one flat export surface, plus the tokens in `rta-ui/style.css`:

```svelte
<script lang="ts">
	import { Breadcrumbs } from 'rta-ui';
	import 'rta-ui/style.css';
</script>
```

## Overview example source

The component above is this file, unchanged:

```svelte
<script lang="ts">
  import {
    Breadcrumbs,
    BreadcrumbsEllipsis,
    BreadcrumbsItem,
    BreadcrumbsLink,
    BreadcrumbsList,
    BreadcrumbsPage,
    BreadcrumbsPath,
    BreadcrumbsSeparator,
  } from 'rta-ui';
</script>

<Breadcrumbs>
  <BreadcrumbsList>
    <BreadcrumbsItem>
      <BreadcrumbsLink href="/">Home</BreadcrumbsLink>
    </BreadcrumbsItem>
    <BreadcrumbsSeparator />
    <BreadcrumbsItem>
      <BreadcrumbsEllipsis />
    </BreadcrumbsItem>
    <BreadcrumbsSeparator />
    <BreadcrumbsItem>
      <BreadcrumbsLink href="/docs">Docs</BreadcrumbsLink>
    </BreadcrumbsItem>
    <BreadcrumbsSeparator />
    <BreadcrumbsItem>
      <BreadcrumbsPage>Components</BreadcrumbsPage>
    </BreadcrumbsItem>
  </BreadcrumbsList>
</Breadcrumbs>

<div class="path">
  <BreadcrumbsPath
    links={[{ href: '/', label: 'Home' }, { href: '/docs', label: 'Docs' }]}
    page="Components"
  />
</div>

<style>
  .path {
    margin-top: 1rem;
  }
</style>
```
